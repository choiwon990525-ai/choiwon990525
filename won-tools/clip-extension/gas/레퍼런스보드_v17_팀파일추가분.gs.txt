const TK_AG_ALIAS = { '오맨': '오멘', 'kj': '킬조이', '킬조': '킬조이', '케이오': '케이/오', 'kayo': '케이/오', '브림': '브림스톤', '데드락': '데드록', '페닉스': '피닉스', '아스': '아스트라' };
function tkAgKey_(s) { const k = tkKey_(s); return tkKey_(TK_AG_ALIAS[k] || k); }
function tkInFolder_(id) {   // 이 파일이 '01 코칭 업무' 폴더 안에 있나
  try { const it = DriveApp.getFileById(id).getParents(); while (it.hasNext()) { if (it.next().getId() === TK_FOLDER) return true; } } catch (e) {}
  return false;
}
