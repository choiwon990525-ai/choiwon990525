function dec_(s) {
  return String(s).replace(/<[^>]*>/g, '').replace(/&amp;/g, '&').replace(/&#0?39;/g, "'").replace(/&quot;/g, '"')
    .replace(/&lt;/g, '<').replace(/&gt;/g, '>').replace(/&ndash;/g, '-').replace(/\s+/g, ' ').trim();
}

/* ───────────── 파싱 ───────────── */
function parseMatch_(html, matchId) {
  const rows = [];
  const patch = (html.match(/Patch ([\d.]+)/) || [])[1] || '';
  const date = (html.match(/data-utc-ts="([^"]+)"/) || [])[1] || '';
  let m;
  const nav = {};
  const reNav = /vm-stats-gamesnav-item[^"]*"\s+data-game-id="(\d+)"[\s\S]*?font-weight: 400;">\s*(\d+)\s*<\/span>/g;
  while ((m = reNav.exec(html))) nav[m[1]] = +m[2];
  const vods = {};
  const reV = /<a href="(https?:\/\/[^"]+)"[^>]*>\s*Map (\d+)\s*<\/a>/g;
  while ((m = reV.exec(html))) vods[+m[2]] = m[1];
  const reG = /<div class="vm-stats-game[^"]*" data-game-id="(\d+|all)"/g;
  const pos = [];
  while ((m = reG.exec(html))) pos.push([m.index, m[1]]);
  let idx = 0;
  for (let i = 0; i < pos.length; i++) {
    if (pos[i][1] === 'all') continue;
    idx++;
    const gid = pos[i][1];
    const chunk = html.slice(pos[i][0], i + 1 < pos.length ? pos[i + 1][0] : html.length);
    const teams = [], scores = [];
    const reT = /<div class="team-name">([\s\S]*?)<\/div>/g;
    while ((m = reT.exec(chunk)) && teams.length < 2) teams.push(dec_(m[1]));
    const reS = /<div class="score[^"]*"[^>]*>\s*(\d+)/g;
    while ((m = reS.exec(chunk)) && scores.length < 2) scores.push(+m[1]);
    const mm = chunk.match(/<div class="map">[\s\S]*?display: inline-block;">\s*([^<]+?)\s*</);
    const map = mm ? dec_(mm[1]).split(' ')[0] : '';
    const tabs = chunk.split('<div class="ovw-table">').slice(1, 3);
    if (teams.length < 2 || scores.length < 2 || tabs.length < 2 || !map) continue;
    const lineups = tabs.map(t => t.split('class="ovw-row"').slice(1).map(r => {
      const n = r.match(/ovw-player-name[^>]*>([^<]*)</);
      const a = r.match(/agents\/[^"]+" alt="[^"]*" title="([^"]+)"/);
      return [n ? dec_(n[1]) : '', a ? agentName_(dec_(a[1])) : ''];
    }).filter(x => x[1]));
    if (lineups[0].length !== 5 || lineups[1].length !== 5) continue;
    const hdr = chunk.slice(0, chunk.indexOf('ovw-table'));
    const mi = hdr.indexOf('<div class="map">');
    const side = seg => ['ct', 't'].map(c => +((seg.match(new RegExp('class="mod-' + c + '">\\s*(\\d+)')) || [])[1] || 0));
    const sd = [side(mi > 0 ? hdr.slice(0, mi) : ''), side(mi > 0 ? hdr.slice(mi) : '')];
    const yt = vods[nav[gid] || idx] || '';
    const vlr = 'https://www.vlr.gg/' + matchId + '/?game=' + gid + '&tab=overview';
    for (const k of [0, 1]) {
      const lu = lineups[k].slice().sort((x, y) => cmp_(x[1], y[1]));
      rows.push([matchId, gid, '', '', map, teams[k], teams[1 - k], scores[k], scores[1 - k],
        lu.map(x => x[1]).join(' / '), lu.map(x => x[0]).join(' / '), yt, vlr, patch, date,
        sd[k][0], sd[k][1], sd[1 - k][0], sd[1 - k][1]]);
    }
  }
  return rows;
}

/* ───────────── 수집 ───────────── */
function updateData() {
  const START = Date.now();
  const dsh = sheet_('_data', true), done = sheet_('_done', true);
  if (dsh.getLastRow() === 0) dsh.appendRow(DATA_HEAD);
  if (done.getLastRow() === 0) done.appendRow(['matchId', 'status', 'checked']);
  const doneMap = {};
  if (done.getLastRow() > 1) done.getRange(2, 1, done.getLastRow() - 1, 3).getValues()
    .forEach((r, i) => doneMap[String(r[0])] = { status: String(r[1]), row: i + 2 });
  const props = PropertiesService.getScriptProperties();
  const evs = CFG.events.filter(e => props.getProperty('ev_' + e[0]) !== 'done');
  const pending = [];
  const evDone = {};
  if (evs.length) {
    const pages = fetchAll_(evs.map(e => 'https://www.vlr.gg/event/matches/' + e[0] + '/?series_id=all'));
    evs.forEach((e, i) => {
      const r = pages[i];
      if (r.getResponseCode() !== 200) { Logger.log('event ' + e[0] + ' HTTP ' + r.getResponseCode()); return; }
      const h = r.getContentText();
      const re = /<a href="\/(\d+)\/[^"]*" class="wf-module-item match-item[\s\S]*?class="ml-status">\s*([^<]*)</g;
      let m, cnt = 0, all = true;
      while ((m = re.exec(h))) {
        cnt++;
        if (m[2].trim() !== 'Completed') { all = false; continue; }
        const d = doneMap[m[1]];
        if (!d) pending.push([m[1], e[1], e[2], false]);
        else if (d.status === 'novod') { all = false; pending.push([m[1], e[1], e[2], true]); }
      }
      evDone[e[0]] = all && cnt > 0;
    });
  }
  let added = 0, ytFixed = 0, more = false;
  const iso = new Date().toISOString();
  for (let i = 0; i < pending.length; i += 15) {
    if (Date.now() - START > 270000) { more = true; break; }
    const batch = pending.slice(i, i + 15);
    const res = fetchAll_(batch.map(p => 'https://www.vlr.gg/' + p[0] + '/'));
    const newRows = [], newDone = [];
    batch.forEach((p, j) => {
      const r = res[j];
      if (r.getResponseCode() !== 200) { more = true; Logger.log('match ' + p[0] + ' HTTP ' + r.getResponseCode()); return; }
      const rows = parseMatch_(r.getContentText(), p[0]);
      rows.forEach(x => { x[2] = p[1]; x[3] = p[2]; });
      const hasVod = rows.every(x => x[11]);
      const old = rows.length && (Date.now() - new Date(String(rows[0][14]).replace(' ', 'T') + 'Z').getTime()) > 3 * 864e5;
      const st = (hasVod || old) ? 'ok' : 'novod';
      if (p[3]) { if (hasVod && rows.length) ytFixed += updateYt_(dsh, rows); }
      else newRows.push(...rows);
      const d = doneMap[p[0]];
      if (d) done.getRange(d.row, 2, 1, 2).setValues([[st, iso]]);
      else newDone.push([p[0], st, iso]);
    });
    if (newDone.length) {
      const start = done.getLastRow() + 1;
      done.getRange(start, 1, newDone.length, 3).setValues(newDone);
      newDone.forEach((d, k) => doneMap[d[0]] = { status: d[1], row: start + k });
    }
    if (newRows.length) {
      dsh.getRange(dsh.getLastRow() + 1, 1, newRows.length, DATA_HEAD.length).setValues(newRows);
      added += newRows.length;
    }
    Utilities.sleep(300);
  }
  if (!more) Object.keys(evDone).forEach(id => { if (evDone[id]) props.setProperty('ev_' + id, 'done'); });
  const out = { added: added, ytFixed: ytFixed, more: more, pending: pending.length, sec: Math.round((Date.now() - START) / 1000) };
  Logger.log(JSON.stringify(out));
  return out;
}

function updateYt_(dsh, rows) {
  const vals = dsh.getRange(2, 1, dsh.getLastRow() - 1, 2).getValues();
  const byGame = {};
  rows.forEach(r => byGame[String(r[1])] = r[11]);
  let n = 0;
  vals.forEach((v, i) => {
    const yt = byGame[String(v[1])];
    if (yt && String(v[0]) === String(rows[0][0])) { dsh.getRange(i + 2, 12).setValue(yt); n++; }
  });
  return n;
}

function rng_(p, col) { return p + '$' + col + '$' + DATA_ROW + ':$' + col; }
function topFormula_(p, st, rg) {
  return '=IFERROR(ARRAYFORMULA(LET(c,' + rng_(p, 'A') + ',t,' + rng_(p, 'B') + ',r,' + rng_(p, 'D') + ',e,' + rng_(p, 'E') + ',g,' + rng_(p, 'F') + ',' +
    'm,(c<>"")*IF(' + st + '="전체",1,e=' + st + ')*IF(' + rg + '="전체",1,g=' + rg + '),' +
    'fc,FILTER(c,m),ft,FILTER(t,m),fr,FILTER(r,m),u,UNIQUE(fc),' +
    'n,MAP(u,LAMBDA(x,SUMPRODUCT(--(fc=x)))),' +
    'tm,MAP(u,LAMBDA(x,COUNTUNIQUE(FILTER(ft,fc=x)))),' +
    'w,MAP(u,LAMBDA(x,SUMPRODUCT((fc=x)*(LEFT(fr,1)="승")))),' +
    'ARRAY_CONSTRAIN(SORT(HSTACK(u,n,tm,w),2,FALSE,3,FALSE,1,TRUE),3,4))),"")';
}
// 맵 탭 오른쪽 '이 맵 조합별 승률' — [조합, 횟수, 팀 수, 종합, 공격, 수비, 주로 쓴 팀]
// 팀 전체일 땐 3번 미만 쓴 조합을 맨 아래로 (팀을 고르면 전부)
function compFormula_(st, rg, tm, so) {
  const C = c => rng_('', c);
  return '=IFERROR(ARRAYFORMULA(LET(c,' + C('A') + ',t,' + C('B') + ',r,' + C('D') + ',e,' + C('E') + ',g,' + C('F') + ',dw,' + C('K') + ',aw,' + C('L') + ',odw,' + C('M') + ',oaw,' + C('N') + ',' +
    'm,(c<>"")*IF(' + st + '="전체",1,e=' + st + ')*IF(' + rg + '="전체",1,g=' + rg + ')*IF(' + tm + '="전체",1,t=' + tm + '),' +
    'u,UNIQUE(FILTER(c,m)),' +
    'n,MAP(u,LAMBDA(x,SUMPRODUCT((c=x)*m))),' +
    'tn,MAP(u,LAMBDA(x,COUNTUNIQUE(FILTER(t,(c=x)*m)))),' +
    'w,MAP(u,LAMBDA(x,SUMPRODUCT((c=x)*m*(LEFT(r,1)="승")))),' +
    'atw,MAP(u,LAMBDA(x,SUMPRODUCT((c=x)*m*aw))),' +
    'atp,MAP(u,LAMBDA(x,SUMPRODUCT((c=x)*m*(aw+odw)))),' +
    'dfw,MAP(u,LAMBDA(x,SUMPRODUCT((c=x)*m*dw))),' +
    'dfp,MAP(u,LAMBDA(x,SUMPRODUCT((c=x)*m*(dw+oaw)))),' +
    'tp,MAP(u,LAMBDA(x,LET(ft,FILTER(t,(c=x)*m),ut,UNIQUE(ft),ct,MAP(ut,LAMBDA(y,SUMPRODUCT(--(ft=y)))),' +
      'q,SORT(HSTACK(ut,ct),2,FALSE,1,TRUE),TEXTJOIN(", ",TRUE,ARRAY_CONSTRAIN(INDEX(q,0,1)&" "&INDEX(q,0,2),3,1))))),' +
    'th,IF(' + tm + '="전체",3,1),wr,w/n,ar,IF(atp=0,-1,atw/atp),dr,IF(dfp=0,-1,dfw/dfp),' +
    'k,IF(' + so + '="횟수",n,IF(' + so + '="공격",ar,IF(' + so + '="수비",dr,wr))),' +
    's,SORT(HSTACK(u,n,tn,wr,IF(ar<0,"",ar),IF(dr<0,"",dr),tp,--(n>=th),k,n),8,FALSE,9,FALSE,10,FALSE,4,FALSE,1,TRUE),' +
    'CHOOSECOLS(s,1,2,3,4,5,6,7))),"")';
}
// 이 맵 자체의 진영 밸런스 (필터 대상 전체의 공격 라운드 승률)
function sideBiasFormula_(st, rg) {
  const C = c => rng_('', c);
  return '=IFERROR(ARRAYFORMULA(LET(t,' + C('B') + ',e,' + C('E') + ',g,' + C('F') + ',aw,' + C('L') + ',odw,' + C('M') + ',' +
    'm,(t<>"")*IF(' + st + '="전체",1,e=' + st + ')*IF(' + rg + '="전체",1,g=' + rg + '),' +
    'v,SUMPRODUCT(m*aw)/SUMPRODUCT(m*(aw+odw)),' +
    '"⚖️ 이 맵 공격 라운드 승률 (필터 대상 전체 평균) "&TEXT(v,"0.0%")&IF(v<0.5,"  → 수비 쪽 유리","  → 공격 쪽 유리"))),"")';
}
function totalFormula_(p, st, rg) {
  return '=ARRAYFORMULA(SUMPRODUCT((' + rng_(p, 'A') + '<>"")*IF(' + st + '="전체",1,' + rng_(p, 'E') + '=' + st + ')*IF(' + rg + '="전체",1,' + rng_(p, 'F') + '=' + rg + ')))';
}
// 한 팀이 이 맵에서 가장 많이 쓴 조합 → [조합, 횟수, 승수, 그 팀 이 맵 전체 경기 수]
function teamFormula_(p, teamCell, st) {
  return '=IF(' + teamCell + '="","",IFERROR(ARRAYFORMULA(LET(c,' + rng_(p, 'A') + ',t,' + rng_(p, 'B') + ',r,' + rng_(p, 'D') + ',e,' + rng_(p, 'E') + ',' +
    'm,(t=' + teamCell + ')*(c<>"")*IF(' + st + '="전체",1,e=' + st + '),' +
    'fc,FILTER(c,m),fr,FILTER(r,m),u,UNIQUE(fc),' +
    'n,MAP(u,LAMBDA(x,SUMPRODUCT(--(fc=x)))),' +
    'w,MAP(u,LAMBDA(x,SUMPRODUCT((fc=x)*(LEFT(fr,1)="승")))),' +
    'HSTACK(ARRAY_CONSTRAIN(SORT(HSTACK(u,n,w),2,FALSE,1,TRUE),1,3),ROWS(fc)))),""))';
}
function rankTeamFormula_(key, k) {
  return '=IFERROR(INDEX(FILTER(\'_순위\'!$B:$D,\'_순위\'!$A:$A=' + key + '),1,' + k + '),"")';
}
function rankLabelFormula_(key) {
  return 'IFERROR(INDEX(FILTER(\'_순위\'!$E:$E,\'_순위\'!$A:$A=' + key + '),1),"")';
}

/* 순위표(_순위 숨김 탭): 세계 1~3위 + 지역별 1~3위. 2026은 챔피언스가 끝나면 챔피언스 최종 순위로 자동 교체 */
function parsePlacements_(h) {
  const i = h.indexOf('Prize Distribution'); if (i < 0) return [];
  const rowsH = h.slice(i).split('<div class="row').slice(2);
  const out = [];
  for (const rh of rowsH) {
    const pl = rh.match(/nowrap;">\s*(\d+)<sup>[a-z]+<\/sup>\s*(–|&ndash;)?/);
    const tm = rh.match(/class="text-of">\s*([^<]+?)\s*</);
    if (!pl || pl[2] || !tm || +pl[1] !== out.length + 1) break;
    out.push(dec_(tm[1]));
    if (out.length === 3) break;
  }
  return out;
}
function ensureRanks_() {
  const R = RANKS[CFG.year]; if (!R) return false;
  let world = R.world.teams, wlabel = R.world.label;
  if (R.champ) {
    try {
      const h = UrlFetchApp.fetch('https://www.vlr.gg/event/' + R.champ + '/playoffs', { muteHttpExceptions: true, headers: UA }).getContentText();
      const got = parsePlacements_(h);
      if (got.length === 3) { world = got; wlabel = '챔피언스 최종 순위'; }
    } catch (e) { Logger.log('챔피언스 순위 확인 실패: ' + e); }
  }
  const rows = [['구분', '1위', '2위', '3위', '기준'], ['세계', world[0] || '', world[1] || '', world[2] || '', wlabel]];
  CFG.regions.forEach(g => { const t = R.region.teams[g] || []; rows.push([g, t[0] || '', t[1] || '', t[2] || '', t.length ? R.region.label : '이 해엔 순위 대회 없음']); });
  const sh = sheet_('_순위', true);
  const old = sh.getLastRow() ? sh.getRange(1, 1, sh.getLastRow(), 5).getDisplayValues() : [];
  const changed = JSON.stringify(old) !== JSON.stringify(rows);
  if (changed) { sh.clear(); sh.getRange(1, 1, rows.length, 5).setValues(rows); }
  return changed;
}

function iconFormula_(cell, k) {
  const a = 'INDEX(SPLIT(' + cell + '," / ",FALSE),1,' + k + ')';
  return '=IF(' + cell + '="","",IFERROR(IMAGE(VLOOKUP(' + a + ',\'_아이콘\'!$A:$B,2,FALSE)),' + a + '))';
}

function ensureIcons_(agents) {
  const sh = sheet_('_아이콘', true);
  const have = {};
  if (sh.getLastRow() > 1) sh.getRange(2, 1, sh.getLastRow() - 1, 2).getValues().forEach(r => have[r[0]] = r[1]);
  if (!Object.keys(have).length) {
    try {
      const j = JSON.parse(UrlFetchApp.fetch('https://valorant-api.com/v1/agents?isPlayableCharacter=true', { muteHttpExceptions: true }).getContentText());
      j.data.forEach(a => have[a.displayName] = a.displayIcon);
    } catch (e) { Logger.log('valorant-api 실패: ' + e); }
  }
  agents.forEach(a => { if (!have[a]) have[a] = 'https://www.vlr.gg/img/vlr/game/agents/' + a.toLowerCase().replace(/[^a-z]/g, '') + '.png'; });
  const rows = Object.keys(have).sort().map(k => [k, have[k]]);
  sh.clear();
  sh.getRange(1, 1, 1, 2).setValues([['agent', 'icon']]);
  sh.getRange(2, 1, rows.length, 2).setValues(rows);
}

function render() {
  const s = ss_();
  const dsh = s.getSheetByName('_data');
  const all = dsh.getRange(2, 1, dsh.getLastRow() - 1, DATA_HEAD.length).getValues();
  const teamRegion = {};
  all.forEach(r => { if (r[3]) teamRegion[r[5]] = r[3]; });
  Object.keys(CFG.teamRegion || {}).forEach(t => { if (!teamRegion[t]) teamRegion[t] = CFG.teamRegion[t]; });
  // 쇼매치처럼 양쪽 팀 모두 리그 소속이 아닌 경기는 제외
  const vals = all.filter(r => r[3] || teamRegion[r[5]] || teamRegion[r[6]]);
  const agents = {};
  vals.forEach(r => String(r[9]).split(' / ').forEach(a => agents[a] = 1));
  ensureIcons_(Object.keys(agents));
  ensureRanks_();
  const maps = CFG.baseMaps.slice();
  vals.forEach(r => { if (maps.indexOf(r[4]) < 0) maps.push(r[4]); });
  maps.sort();
  const stamp = nowKst_();
  maps.forEach(mp => renderMap_(s, mp, vals.filter(r => r[4] === mp), teamRegion, stamp));
  renderSummary_(s, maps, stamp);
  renderTeamStats_(s, vals, teamRegion, stamp);
  const champ = renderChampions_(s, vals, stamp);
  // 탭 순서: 요약 → (챔피언스) → 팀 승률 → 맵 알파벳
  ['요약'].concat(champ ? [CHAMP_TAB] : [], ['팀 승률'], maps).forEach((n, i) => { s.setActiveSheet(s.getSheetByName(n)); s.moveActiveSheet(i + 1); });
  if (CFG.year !== 2026) {
    const keep = ['요약', '팀 승률', CHAMP_TAB, '_data', '_done', '_아이콘', '_순위', '레퍼런스 보드', '_설정'].concat(maps);
    s.getSheets().forEach(x => { if (keep.indexOf(x.getName()) < 0) s.deleteSheet(x); });
  }
  s.setActiveSheet(s.getSheetByName('요약'));
  PropertiesService.getScriptProperties().setProperty(CFG.year === 2026 ? 'rendered' : 'rendered_' + CFG.year, stamp);
}

/* 팀 승률 탭: 시즌 전체 종합(맵) · 공격 라운드 · 수비 라운드 승률을 각각 높은 순으로 */
function renderTeamStats_(s, vals, teamRegion, stamp) {
  let sh = s.getSheetByName('팀 승률');
  if (!sh) sh = s.insertSheet('팀 승률');
  resetSheet_(sh);
  const T = {};
  vals.forEach(r => {
    const t = String(r[5]);
    const x = T[t] = T[t] || { team: t, region: String(teamRegion[t] || r[3] || '국제'), mw: 0, mp: 0, aw: 0, ap: 0, dw: 0, dp: 0 };
    x.mp++; if (+r[7] > +r[8]) x.mw++;
    const dW = +r[15] || 0, aW = +r[16] || 0, odW = +r[17] || 0, oaW = +r[18] || 0;
    if (dW + aW + odW + oaW > 0) { x.aw += aW; x.ap += aW + odW; x.dw += dW; x.dp += dW + oaW; }
  });
  const teams = Object.keys(T).map(k => T[k]);
  // 표본 적은 팀(승강·게스트 팀 등)이 위로 튀지 않게: 맵 수가 중앙값의 40% (최소 10) 미만이면 회색 + 맨 아래
  const mps = teams.map(x => x.mp).sort((a, c) => a - c);
  const TH = Math.max(10, Math.round((mps[Math.floor(mps.length / 2)] || 0) * 0.4));
  const blocks = [
    { title: '🏆 종합 승률 (맵 기준)', rate: x => x.mp ? x.mw / x.mp : 0, n: x => x.mp, sample: x => x.mw + '승 ' + (x.mp - x.mw) + '패' },
    { title: '⚔️ 공격 라운드 승률', rate: x => x.ap ? x.aw / x.ap : 0, n: x => x.ap, sample: x => x.aw + ' / ' + x.ap + ' 라운드' },
    { title: '🛡️ 수비 라운드 승률', rate: x => x.dp ? x.dw / x.dp : 0, n: x => x.dp, sample: x => x.dw + ' / ' + x.dp + ' 라운드' }
  ];
  const W = 6; // 블록 폭 (5칸 + 여백 1칸)
  const need = 4 + teams.length + 3;
  if (sh.getMaxRows() < need) sh.insertRowsAfter(sh.getMaxRows(), need - sh.getMaxRows());
  if (sh.getMaxColumns() < W * 3) sh.insertColumnsAfter(sh.getMaxColumns(), W * 3 - sh.getMaxColumns());
  sh.getRange('A1').setValue('팀별 시즌 승률 · ' + CFG.label + ' (VLR.gg)').setFontWeight('bold').setFontSize(12);
  sh.getRange('A2').setValue('종합 = 맵 승률 · 공격/수비 = 그 진영에서 딴 라운드 비율 (연장전 라운드 제외) · 시즌 맵 ' + TH + '개 미만 팀은 표본이 적어 회색으로 맨 아래 · 갱신 ' + stamp)
    .setFontColor('#6b7280').setFontSize(9);
  blocks.forEach((b, bi) => {
    const c0 = 1 + bi * W;
    const list = teams.slice().sort((a, c) => (a.mp < TH) - (c.mp < TH) || b.rate(c) - b.rate(a) || b.n(c) - b.n(a) || cmp_(a.team, c.team));
    sh.getRange(3, c0, 1, 5).merge().setValue(b.title).setFontWeight('bold').setBackground('#1f2937').setFontColor('#ffffff').setHorizontalAlignment('center');
    sh.getRange(4, c0, 1, 5).setValues([['순위', '팀', '지역', '승률', '표본']]).setFontWeight('bold').setBackground('#e5e7eb').setHorizontalAlignment('center');
    if (!list.length) return;
    sh.getRange(5, c0, list.length, 5).setValues(list.map((x, i) => [x.mp < TH ? '–' : i + 1, x.team, x.region, b.rate(x), b.sample(x)]));
    sh.getRange(5, c0 + 3, list.length, 1).setNumberFormat('0.0%').setFontWeight('bold');
    sh.getRange(5, c0, list.length, 1).setHorizontalAlignment('center');
    sh.getRange(5, c0 + 2, list.length, 3).setHorizontalAlignment('center');
    sh.getRange(5, c0, list.length, 5).setFontColors(list.map(x => new Array(5).fill(x.mp < TH ? '#9ca3af' : '#111827')));
    sh.getRange(5, c0, Math.min(3, list.length), 5).setBackground('#fef3c7');
    const rule = SpreadsheetApp.newConditionalFormatRule().setGradientMaxpoint('#86efac').setGradientMidpointWithValue('#ffffff', SpreadsheetApp.InterpolationType.PERCENT, '50')
      .setGradientMinpoint('#fca5a5').setRanges([sh.getRange(5, c0 + 3, list.length, 1)]).build();
    sh.setConditionalFormatRules(sh.getConditionalFormatRules().concat([rule]));
    sh.setColumnWidth(c0, 40); sh.setColumnWidth(c0 + 1, 150); sh.setColumnWidth(c0 + 2, 80);
    sh.setColumnWidth(c0 + 3, 70); sh.setColumnWidth(c0 + 4, 110);
    if (bi < 2) sh.setColumnWidth(c0 + 5, 18);
  });
  sh.setFrozenRows(4);
}

/* 이미 모아둔 경기의 진영별 라운드(공/수)를 VLR에서 다시 읽어 채움. 다 채우면 true */
function backfillSides_() {
  const START = Date.now();
  const dsh = ss_().getSheetByName('_data');
  dsh.getRange(1, 1, 1, DATA_HEAD.length).setValues([DATA_HEAD]);
  const n = dsh.getLastRow() - 1;
  const vals = dsh.getRange(2, 1, n, DATA_HEAD.length).getValues();
  const need = [...new Set(vals.filter(r => r[15] === '').map(r => String(r[0])))];
  let more = false;
  for (let i = 0; i < need.length; i += 20) {
    if (Date.now() - START > 200000) { more = true; break; }
    const batch = need.slice(i, i + 20);
    const res = fetchAll_(batch.map(id => 'https://www.vlr.gg/' + id + '/'));
    batch.forEach((id, j) => {
      if (res[j].getResponseCode() !== 200) { more = true; return; }
      const by = {};
      parseMatch_(res[j].getContentText(), id).forEach(r => by[String(r[1]) + '|' + r[5]] = r.slice(15, 19));
      vals.forEach(r => {
        if (String(r[0]) === id && r[15] === '') { const v = by[String(r[1]) + '|' + r[5]] || [0, 0, 0, 0]; for (let q = 0; q < 4; q++) r[15 + q] = v[q]; }
      });
    });
    Utilities.sleep(200);
  }
  dsh.getRange(2, 16, n, 4).setValues(vals.map(r => r.slice(15, 19)));
  Logger.log(CFG.year + ' 진영 라운드 채움: 남은 경기 ' + (more ? '있음' : '없음'));
  return !more;
}

/* ───────────── 실행 / 트리거 / 검증 ───────────── */
function run() {
  useYear_(2026);
  // 연도별 후속 작업(buildPending)이 끊겨 있으면 다시 이어 붙임
  const pp = PropertiesService.getScriptProperties();
  if ([2026, 2025, 2024, 2023].some(y => !pp.getProperty('sides6_' + y)) &&
      !ScriptApp.getProjectTriggers().some(t => t.getHandlerFunction() === 'buildPending'))
    ScriptApp.newTrigger('buildPending').timeBased().after(60 * 1000).create();
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(1000)) return;
  try {
    const r = updateData();
    const rendered = PropertiesService.getScriptProperties().getProperty('rendered');
    const rankChanged = ensureRanks_();
    // 다시 그리다 시간 초과로 끊기면 dirty 표시가 남아 다음 시간에 다시 그림
    const dirty = PropertiesService.getScriptProperties().getProperty('dirty');
    if (!r.more && (r.added > 0 || r.ytFixed > 0 || !rendered || rankChanged || dirty)) {
      PropertiesService.getScriptProperties().setProperty('dirty', nowKst_());
      render();
      PropertiesService.getScriptProperties().deleteProperty('dirty');
    }
  } finally { lock.releaseLock(); }
}

function setupTrigger() {
  ScriptApp.getProjectTriggers().forEach(t => { if (t.getHandlerFunction() === 'run') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('run').timeBased().everyHours(1).create();
  Logger.log('매시간 run 트리거 등록');
  buildPending();
}

/* 2025·2024·2023 시트를 한 해씩 만들고, 남은 해는 1분 뒤 자동으로 이어서 만듦 (다 만들면 멈춤) */
function buildPending() {
  ScriptApp.getProjectTriggers().forEach(t => { if (t.getHandlerFunction() === 'buildPending') ScriptApp.deleteTrigger(t); });
  const props = PropertiesService.getScriptProperties();
  const years = [2025, 2024, 2023];
  const todo = years.filter(y => !props.getProperty('rendered_' + y));
  if (!todo.length) {
    const rr = [2026, 2025, 2024, 2023].filter(y => !props.getProperty('rr3_' + y));
    if (rr.length) {
      useYear_(rr[0]); render(); props.setProperty('rr3_' + rr[0], nowKst_()); useYear_(2026);
      Logger.log(rr[0] + ' 다시 그림 (상위 팀 조합 추가)');
      if (rr.length > 1) ScriptApp.newTrigger('buildPending').timeBased().after(60 * 1000).create();
      return;
    }
    const sd = [2026, 2025, 2024, 2023].filter(y => !props.getProperty('sides6_' + y));
    if (sd.length) {
      // 다음 실행을 먼저 예약 (이번 실행이 시간 초과로 끊겨도 이어지게). 겹치면 잠금으로 건너뜀
      ScriptApp.newTrigger('buildPending').timeBased().after(60 * 1000).create();
      const lock = LockService.getScriptLock();
      if (!lock.tryLock(5000)) return;
      const y = sd[0];
      try {
        useYear_(y);
        if (!props.getProperty('sidesfill_' + y)) {
          if (backfillSides_()) props.setProperty('sidesfill_' + y, nowKst_());
        } else {
          const tries = +(props.getProperty('sidestry6_' + y) || 0) + 1;
          props.setProperty('sidestry6_' + y, String(tries));
          if (tries > 3) { props.setProperty('sides6_' + y, 'skip'); Logger.log(y + ' 다시 그리기 3번 실패 — 건너뜀'); }
          else { render(); props.setProperty('sides6_' + y, nowKst_()); Logger.log(y + ' 팀 승률 탭 완료'); }
        }
      } finally { lock.releaseLock(); useYear_(2026); }
      return;
    }
    Logger.log('연도별 시트 모두 완료');
    return;
  }
  buildYear_(todo[0]);
  ScriptApp.newTrigger('buildPending').timeBased().after(60 * 1000).create();
}

function verify() {
  const s = ss_();
  const vals = s.getSheetByName('_data').getRange(2, 1, s.getSheetByName('_data').getLastRow() - 1, DATA_HEAD.length).getValues();
  const report = [];
  const maps = {};
  vals.forEach(r => (maps[r[4]] = maps[r[4]] || []).push(r));
  Object.keys(maps).sort().forEach(mp => {
    const rows = maps[mp], cnt = {}, tms = {}, win = {};
    rows.forEach(r => { cnt[r[9]] = (cnt[r[9]] || 0) + 1; (tms[r[9]] = tms[r[9]] || {})[r[5]] = 1; win[r[9]] = (win[r[9]] || 0) + (+r[7] > +r[8] ? 1 : 0); });
    const exp = Object.keys(cnt).sort((a, b) => cnt[b] - cnt[a] || Object.keys(tms[b]).length - Object.keys(tms[a]).length || cmp_(a.toLowerCase(), b.toLowerCase()))
      .slice(0, 3).map(c => [c, cnt[c], Object.keys(tms[c]).length, win[c]].join('|'));
    const sh = s.getSheetByName(mp);
    sh.getRange('C1').setValue('전체'); sh.getRange('E1').setValue('전체');
    SpreadsheetApp.flush();
    const got = sh.getRange('K4:N6').getDisplayValues().map(r => r.join('|'));
    const recRows = sh.getLastRow() - DATA_ROW + 1;
    const ok = JSON.stringify(exp) === JSON.stringify(got) && recRows === rows.length;
    report.push(mp + ' ' + (ok ? 'OK' : 'MISMATCH') + ' rows=' + recRows + '/' + rows.length + (ok ? '' : ' exp=' + JSON.stringify(exp) + ' got=' + JSON.stringify(got)));
  });
  Logger.log(report.join('\n'));
  return report;
}
