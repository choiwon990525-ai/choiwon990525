/* ======================= v16: 그림 M열 (2026-10-09) =======================
   코치(10-09): '시트로 옮길 때 사진을 M라인에 맞춰서' → 새 그림은 M열 선(옆 0px)에 · 위→아래 차례는 그대로
   예전에 C열에 넣은 그림은 확장이 tacticLinks items[].move로 한 번 옮김 — C열·옆 12px·대체 글 'R13 · …'인 그림만(코치가 옮기거나 직접 넣은 그림은 그대로) */
function tkMoveImgs_(sh) {
  let k = 0;
  sh.getImages().forEach(function (im) {
    try {
      const a = im.getAnchorCell();
      if (a.getColumn() !== 3 || im.getAnchorCellXOffset() !== 12) return;
      if (!/^R\d{1,2} · /.test(String(im.getAltTextTitle() || ''))) return;
      const oy = im.getAnchorCellYOffset();
      im.setAnchorCell(sh.getRange(a.getRow(), TK_IMG_COL)); im.setAnchorCellXOffset(TK_IMG_X); im.setAnchorCellYOffset(oy); k++;
    } catch (e) {}
  });
  return k;
}
