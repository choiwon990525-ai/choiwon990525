function tkSetLink_(sh, url, text) {   // 제목 오른쪽 C1: '📊 정리 슬라이드 R13 ↗' (누르면 PPT 그 라운드 장)
  if (!url) return;
  const t = String(text || '📊 정리 슬라이드 ↗');
  const rt = SpreadsheetApp.newRichTextValue().setText(t).setLinkUrl(String(url)).build();
  const c = sh.getRange('C1'); c.setRichTextValue(rt); c.setFontSize(10).setFontColor('#4A88C7').setVerticalAlignment('middle');
}
function tacticLinks_(d) {
  const ss = tkOpen_(d);
  if (!ss) return { ok: false, err: '텍틱 파일을 못 열었어요' };
  const lock = LockService.getScriptLock(); lock.waitLock(30000);
  try {
    const byId = {}; ss.getSheets().forEach(function (s) { byId[String(s.getSheetId())] = s; });
    let n = 0; const miss = [];
    (d.items || []).forEach(function (it) {
      const sh = (it.sheetId != null && byId[String(it.sheetId)]) || (it.tab ? ss.getSheetByName(String(it.tab)) : null);
      if (!sh || !it.url) { miss.push(it.tab || it.sheetId); return; }
      tkSetLink_(sh, it.url, it.text); n++;
    });
    let deck = null;
    if (d.deck) { try { deck = tkPutDeck_(ss, d.deck); } catch (e) { deck = { added: false, err: String(e && e.message || e) }; } }
    SpreadsheetApp.flush();
    return { ok: true, n: n, miss: miss, deck: deck, file: ss.getName(), fileId: ss.getId() };
  } finally { lock.releaseLock(); }
}
function tkDeckId_(u) { const m = /\/presentation\/d\/([\w-]{20,})/.exec(String(u || '')); return m ? m[1] : (/^[\w-]{20,}$/.test(String(u || '')) ? String(u) : ''); }
function tkDataTab_(ss) {   // '자료' 탭 — 없으면 '예시'(없으면 '템플릿') 바로 뒤에 새로
  let sh = ss.getSheetByName('자료'); if (sh) return sh;
  const after = ss.getSheetByName('예시') || ss.getSheetByName('템플릿');
  sh = ss.insertSheet('자료', after ? after.getIndex() : ss.getSheets().length);
  try { sh.setColumnWidth(2, 640); } catch (e) {}
  return sh;
}
function tkPutDeck_(ss, deck) {   // 자료 탭 B열 마지막 글 다음 줄에 '📊 PPT 이름'(링크) — 탭 어디든 같은 PPT 주소·링크가 있으면 안 넣음
  const id = tkDeckId_(deck); if (!id) return { added: false, err: '정리 슬라이드 주소가 없어요' };
  const sh = tkDataTab_(ss), rg = sh.getDataRange(), vals = rg.getValues(), rts = rg.getRichTextValues(), fms = rg.getFormulas();
  let has = false, last = 1;
  vals.forEach(function (row, r) {
    row.forEach(function (v, c) {
      const s = String(v == null ? '' : v); if (c === 1 && s.trim()) last = Math.max(last, r + 1);
      if (s.indexOf(id) >= 0 || String((fms[r] && fms[r][c]) || '').indexOf(id) >= 0) has = true;   // 글자·HYPERLINK 수식
      const rt = rts[r] && rts[r][c]; if (rt && !has) rt.getRuns().forEach(function (x) { const u = x.getLinkUrl(); if (u && u.indexOf(id) >= 0) has = true; });
    });
  });
  if (has) return { added: false, tab: sh.getName() };
  let title = '정리 슬라이드'; try { title = DriveApp.getFileById(id).getName(); } catch (e) {}
  const row = last + 1, rt = SpreadsheetApp.newRichTextValue().setText('📊 ' + title).setLinkUrl('https://docs.google.com/presentation/d/' + id + '/edit').build();
  sh.getRange(row, 2).setRichTextValue(rt).setFontSize(11);
  return { added: true, tab: sh.getName(), row: row };
}
function tkSlideIds_(deckId) {
  const out = {};
  if (!deckId) return out;
  SlidesApp.openById(deckId).getSlides().forEach(function (sl) {
    let n = null;
    sl.getShapes().some(function (sh) { let t = ''; try { t = sh.getText().asString().trim(); } catch (e) {} const m = /^R(\d{1,2})$/.exec(t); if (m) { n = +m[1]; return true; } return false; });
    if (n != null && !out[n]) out[n] = sl.getObjectId();
  });
  return out;
}
function deckSlides_(d) {
  const id = tkDeckId_(d.deck);
  if (!id) return { ok: false, err: '정리 슬라이드 주소가 없어요' };
  try { return { ok: true, deck: id, slides: tkSlideIds_(id) }; } catch (e) { return { ok: false, err: '정리 슬라이드를 못 열었어요: ' + e.message }; }
}
